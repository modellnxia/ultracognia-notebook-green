        self.fx = exchange_rate  # EUR/BRL
        self.cost_br = cost_br_monthly
        self.hours_billed = 160

    def calculate_gross_margin(self):
        revenue_brl = (self.rate_eur * self.hours_billed) * self.fx
        margin_abs = revenue_brl - self.cost_br
        margin_pct = (margin_abs / revenue_brl) * 100
        
        return {
            "revenue_total": round(revenue_brl, 2),
            "margin_absolute": round(margin_abs, 2),
            "margin_percentage": f"{round(margin_pct, 2)}%"
        }
